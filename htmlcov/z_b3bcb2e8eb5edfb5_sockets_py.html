1"""Pure parsers for the socket-view channels (spec §4). 

2 

3Text in, typed values out, no I/O. Only the /proc/net/{tcp,udp}[6] tables are 

4parsed -- they share a column layout, and they carry the detail the socket 

5detector needs. (The inode-only families and the "held fd in no table" signal 

6were dropped after the phase-4a step-0 calibration; see docs/limitations.md L25.) 

7Each case mirrors docs/step0-phase4/. 

8""" 

9from __future__ import annotations 

10 

11import socket 

12import struct 

13from dataclasses import dataclass 

14 

15 

16class SocketParseError(ValueError): 

17 """A /proc/net socket table could not be parsed.""" 

18 

19 

20@dataclass(frozen=True) 

21class NetRow: 

22 inode: int 

23 kind: str # "tcp" | "tcp6" | "udp" | "udp6" 

24 state: str | None 

25 local: str | None 

26 remote: str | None 

27 uid: int | None 

28 

29 

30#: TCP state codes (hex) -> name. UDP reuses the column; unknown codes pass 

31#: through as the raw hex so nothing is silently mislabelled. 

32_STATES = { 

33 "01": "ESTABLISHED", "02": "SYN_SENT", "03": "SYN_RECV", "04": "FIN_WAIT1", 

34 "05": "FIN_WAIT2", "06": "TIME_WAIT", "07": "CLOSE", "08": "CLOSE_WAIT", 

35 "09": "LAST_ACK", "0A": "LISTEN", "0B": "CLOSING", "0C": "NEW_SYN_RECV", 

36} 

37 

38 

39def _decode_addr(hexaddr: str) -> str: 

40 """'0100007F:0035' -> '127.0.0.1:53'. v4 = 8 hex (little-endian word), 

41 v6 = 32 hex (four little-endian 32-bit words).""" 

42 ip_hex, _, port_hex = hexaddr.partition(":") 

43 port = int(port_hex, 16) 

44 if len(ip_hex) == 8: 

45 packed = struct.pack("<I", int(ip_hex, 16)) 

46 ip = socket.inet_ntop(socket.AF_INET, packed) 

47 elif len(ip_hex) == 32: 

48 raw = bytes.fromhex(ip_hex) 

49 packed = b"".join(raw[i:i + 4][::-1] for i in range(0, 16, 4)) 

50 ip = socket.inet_ntop(socket.AF_INET6, packed) 

51 else: 

52 raise SocketParseError(f"bad address field: {hexaddr!r}") 

53 return f"{ip}:{port}" 

54 

55 

56def parse_net_tcp(text: str, kind: str) -> list[NetRow]: 

57 """Parse a /proc/net/{tcp,tcp6,udp,udp6} table into NetRows.""" 

58 rows: list[NetRow] = [] 

59 for line in text.splitlines(): 

60 parts = line.split() 

61 if not parts or parts[0] == "sl" or not parts[0].endswith(":"): 

62 continue # header or blank 

63 if len(parts) < 10: 

64 raise SocketParseError(f"short {kind} row: {line!r}") 

65 try: 

66 rows.append(NetRow( 

67 inode=int(parts[9]), kind=kind, 

68 state=_STATES.get(parts[3].upper(), parts[3]), 

69 local=_decode_addr(parts[1]), remote=_decode_addr(parts[2]), 

70 uid=int(parts[7]), 

71 )) 

72 except (ValueError, IndexError) as exc: 

73 raise SocketParseError(f"bad {kind} row: {line!r}") from exc 

74 return rows
