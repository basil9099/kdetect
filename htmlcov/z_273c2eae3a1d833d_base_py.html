1"""Abstract interfaces for the collector layer. 

2 

3Spec section 4.1 splits collection into three layers, and only the Source 

4touches the operating system: 

5 

6 Source -> Parse -> Collector 

7 (all I/O) (pure functions) (assembles Observation) 

8 

9Collectors receive their source as an argument, so a fixture-driven test runs 

10the identical code path as a live capture (principle P3). There is no 

11`if testing:` branch anywhere in this package. 

12""" 

13 

14from __future__ import annotations 

15 

16from abc import ABC, abstractmethod

18from kdetect.models import Observation, TrustLevel

21class ProcSourceError(Exception): 

22 """Base for every failure a ProcSource can report. 

23 

24 Implementations translate OSError into one of the subclasses below. No raw 

25 OSError may escape a ProcSource - that translation is the entire reason 

26 this layer exists, and it is what lets a fixture replay a failure that 

27 never actually happened. 

28 """ 

29 

30 

31class Vanished(ProcSourceError): 

32 """ESRCH/ENOENT - the entity is gone, or never had this attribute. 

33 

34 Normal on a live system: processes exit while you are walking /proc. Also 

35 what a kernel thread's `exe` gives, since it genuinely has no executable. 

36 Does NOT degrade an Observation's status (spec section 4.2). 

37 """ 

38 

39 

40class Denied(ProcSourceError): 

41 """EACCES/EPERM - it exists, we are not permitted to read it. 

42 

43 Materially different from Vanished. As root this is anomalous rather than 

44 routine, but the source does not judge that; it records the fact and lets 

45 analysis weigh it against capture.euid (principle P1). 

46 """ 

47 

48 

49class Unreadable(ProcSourceError): 

50 """Any other I/O failure.""" 

51

53class ProcSource(ABC): 

54 """Somewhere process information can be read from. 

55 

56 Deliberately tiny: three operations. Two implementations exist - 

57 LiveProcSource reads /proc, FixtureProcSource replays a captured tree. 

58 """ 

59 

60 @abstractmethod 

61 def list_pids(self) -> list[int]: 

62 """Every pid this source knows about, unique and sorted ascending.""" 

63 

64 @abstractmethod 

65 def read_text(self, pid: int, name: str) -> str: 

66 """Read /proc/<pid>/<name> as text. e.g. "stat", "cmdline", "status".""" 

67 

68 @abstractmethod 

69 def read_link(self, pid: int, name: str) -> str: 

70 """Resolve the symlink /proc/<pid>/<name>. e.g. "exe", "cwd". 

71 

72 The target is returned verbatim, including any " (deleted)" suffix - 

73 a running binary unlinked from disk is one of the strongest single 

74 indicators available, and stripping it would discard the finding. 

75 """ 

76

78class SignalSource(ABC): 

79 """The process-existence channel that does not go through readdir (spec §4). 

80 

81 kill(id, 0) asks the kernel whether an id exists; /proc/<id>/status is read 

82 by path lookup, not getdents. Both bypass a directory-listing hook, which 

83 is why this is a MEDIUM channel independent of procfs. 

84 """ 

85 

86 @abstractmethod 

87 def pid_max(self) -> int: ... 

88 

89 @abstractmethod 

90 def sweep(self, pid_max: int) -> list[int]: 

91 """Every id in 1..pid_max that exists, sorted. EPERM counts as exists.""" 

92 

93 @abstractmethod 

94 def read_status(self, task_id: int) -> tuple[int, str | None] | None: 

95 """(Tgid, Name) from /proc/<id>/status, or None if it could not be read."""

98class ModuleSource(ABC): 

99 """The module-view channels (spec §5). /proc/modules is the LOW listing a 

100 rootkit unlinks itself from; tainted, vmallocinfo and ftrace are channels 

101 that do NOT share that list's source. A channel that cannot be read returns 

102 None, and the differ skips it rather than treating absence as agreement.""" 

103 

104 @abstractmethod 

105 def read_proc_modules(self) -> str: ... 

106 @abstractmethod 

107 def read_tainted(self) -> int: ... 

108 @abstractmethod 

109 def read_vmallocinfo(self) -> str | None: ... 

110 @abstractmethod 

111 def read_ftrace_functions(self) -> str | None: ...

114class KernelHookSource(ABC): 

115 """The hook-surface channels (spec §4). enabled_functions and kprobes are

117 a module. A channel that cannot be read returns None, and the collector 

118 records its absence rather than treating it as agreement."""

128class SocketSource(ABC): 

129 """The socket-view channels (spec §4): the /proc/net/* tables and per-process

131 fds return {} (a hidden pid contributes nothing, not an error)."""

139class Collector(ABC): 

140 """Produces one Observation - one whole view of one kind of entity.""" 

141 

142 name: str 

143 view: str 

144 trust_level: TrustLevel

147 @abstractmethod

150
