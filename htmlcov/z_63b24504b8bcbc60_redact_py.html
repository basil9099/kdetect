1"""Snapshot redaction for sharing (limitation L13). 

2 

3/proc/[pid]/cmdline is captured verbatim, so a snapshot can carry credentials a 

4process was handed as arguments. Cross-view findings never depend on cmdline 

5(they key on pids/tgids/module/socket data), so replacing every process cmdline 

6wholesale removes the secrets without changing any conclusion. Reports do not 

7carry cmdline at all (spec §7); this is for scrubbing a *snapshot* to share. 

8""" 

9from __future__ import annotations 

10 

11_PLACEHOLDER = ["[redacted]"] 

12 

13 

14def redact_snapshot(snap: dict) -> dict: 

15 """Replace every process entity's cmdline with a placeholder, in place.""" 

16 for obs in snap.get("observations", []): 

17 for entity in obs.get("entities", {}).values(): 

18 if isinstance(entity, dict) and "cmdline" in entity: 

19 entity["cmdline"] = list(_PLACEHOLDER) 

20 return snap
